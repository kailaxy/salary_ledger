cmake_minimum_required(VERSION 3.22.1)
project(ledger_ai_bench LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK compile_pool)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(RUNTIME_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp")
if(NOT EXISTS "${RUNTIME_DIR}/include/llama.h")
 message(FATAL_ERROR "Run android-ai-benchmark/tools/fetch-runtime.sh before building")
endif()
execute_process(COMMAND git -C "${RUNTIME_DIR}" rev-parse HEAD OUTPUT_VARIABLE RUNTIME_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT RUNTIME_REV STREQUAL "74d4f5b041ad837153b0e90fc864b8290e01d8d5")
 message(FATAL_ERROR "Unexpected llama.cpp revision; run tools/fetch-runtime.sh")
endif()
execute_process(COMMAND git -C "${RUNTIME_DIR}" status --porcelain --untracked-files=no OUTPUT_VARIABLE RUNTIME_DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE)
if(RUNTIME_DIRTY)
 message(FATAL_ERROR "Runtime source has tracked changes")
endif()
add_subdirectory("${RUNTIME_DIR}" "${CMAKE_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(ledgerbench SHARED engine.cpp jni.cpp)
target_include_directories(ledgerbench PRIVATE "${RUNTIME_DIR}/include")
target_link_libraries(ledgerbench PRIVATE llama log android)
target_link_options(ledgerbench PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
